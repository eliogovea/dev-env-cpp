1/*
2 * The program executes a command in a child process, redirects its output and
3 * error to pipes, and handles timeouts and signals in the parent process.
4 *
5 * Command-line arguments:
6 * - [--timeout|-t TIMEOUT]: Optional flag to specify a timeout in seconds.
7 * Defaults to 10 seconds if not provided.
8 * - COMMAND [ARGS...]: The command to execute and its arguments. This must be
9 * provided.
10 *
11 * The program will:
12 * 1. Set up pipes for stdout and stderr.
13 * 2. Fork a child process.
14 * 3. In the child process, redirect stdout and stderr to the pipes and execute the command.
15 * 4. In the parent process, use `poll` to monitor pipes and handle timeouts and signals.
16 * 5. Terminate the child process if it does not complete within the specified timeout.
17 */

48// Function to create and configure a timerfd
49int create_timer_fd(int timeout)
50{
51 // Create a timer file descriptor with CLOCK_MONOTONIC and non-blocking mode
52 // CLOCK_MONOTONIC ensures that the timer is not affected by system clock changes
53 int timer_fd = timerfd_create(CLOCK_MONOTONIC, TFD_NONBLOCK);
54 if (timer_fd == -1) {
55 perror("timerfd_create failed");
56 exit(EXIT_FAILURE);
57 }
58
59 // Configure the timer to expire after the specified timeout
60 struct itimerspec timer_spec;
61 timer_spec.it_interval.tv_sec = 0; // No periodic timer
62 timer_spec.it_interval.tv_nsec = 0; //
63 timer_spec.it_value.tv_sec = timeout; // Set initial expiration time
64 timer_spec.it_value.tv_nsec = 0; //
65
66 // Set the timer with the specified timeout value
67 if (timerfd_settime(timer_fd, 0, &timer_spec, NULL) == -1) {
68 perror("timerfd_settime failed");
69 close(timer_fd);
70 exit(EXIT_FAILURE);
71 }
72
73 return timer_fd;
74}

76// Function to create a signalfd for handling SIGCHLD signals
77int create_signalfd(void)
78{
79 sigset_t sigset;
80
81 // Initialize an empty signal set
82 sigemptyset(&sigset);
83
84 // Add SIGCHLD to the signal set to handle child process termination
85 sigaddset(&sigset, SIGCHLD);
86
87 // Block SIGCHLD signals in the current process so they can be handled via signalfd
88 if (sigprocmask(SIG_BLOCK, &sigset, NULL) == -1) {
89 perror("sigprocmask failed");
90 exit(EXIT_FAILURE);
91 }
92
93 // Create a file descriptor for handling signals with non-blocking mode
94 int signal_fd = signalfd(-1, &sigset, SFD_NONBLOCK);
95 if (signal_fd == -1) {
96 perror("signalfd_create failed");
97 exit(EXIT_FAILURE);
98 }
99
100 return signal_fd;
101}

103// Function to set a file descriptor to non-blocking mode
104void set_fd_nonblocking(int fd)
105{
106 // Get the current file descriptor flags
107 int flags = fcntl(fd, F_GETFL, 0);
108 if (flags == -1) {
109 perror("fcntl F_GETFL failed");
110 exit(EXIT_FAILURE);
111 }
112 // Set the file descriptor to non-blocking mode
113 if (fcntl(fd, F_SETFL, flags | O_NONBLOCK) == -1) {
114 perror("fcntl F_SETFL failed");
115 exit(EXIT_FAILURE);
116 }
117}

119int main(int argc, char* argv[], char* envp[])
120{
121 // Parse command-line arguments for the help option
122 if (argc > 1 && (strcmp(argv[1], "--help") == 0 || strcmp(argv[1], "-h") == 0)) {
123 print_usage(argv[0]);
124 return EXIT_SUCCESS; // Exit after printing usage information
125 }
126
127 int timeout = 10; // Default timeout value in seconds
128 int command_index = 1; // Index of the command in argv
129
130 // Parse command-line arguments for the timeout option
131 if (argc > 1 && (strcmp(argv[1], "--timeout") == 0 || strcmp(argv[1], "-t") == 0)) {
132 // Print error if timeout value is missing
133 if (argc < 3) {
134 fprintf(stderr, "Error: Timeout value is required\n");
135 print_usage(argv[0]);
136 return EXIT_FAILURE;
137 }
138 // Convert timeout value from string to integer
139 timeout = atoi(argv[2]);
140 if (timeout <= 0) {
141 fprintf(stderr, "Error: Timeout must be a positive integer\n");
142 print_usage(argv[0]);
143 return EXIT_FAILURE;
144 }
145 // Update command index if timeout option is present
146 command_index = 3;
147 }
148
149 // Ensure a command is provided
150 if (argc <= command_index) {
151 fprintf(stderr, "Error: No command provided\n");
152 print_usage(argv[0]);
153 return EXIT_FAILURE;
154 }
155
156 // Create pipes for stdout and stderr
157 int stdout_pipe[2];
158 int stderr_pipe[2];
159 if (pipe2(stdout_pipe, O_CLOEXEC) == -1 || pipe2(stderr_pipe, O_CLOEXEC) == -1) {
160 perror("pipe creation failed");
161 return EXIT_FAILURE;
162 }
163
164 // Set pipe read ends to non-blocking mode to handle I/O efficiently
165 set_fd_nonblocking(stdout_pipe[0]);
166 set_fd_nonblocking(stderr_pipe[0]);
167
168 // Fork the process to create a child
169 pid_t child_pid = fork();
170 if (child_pid == -1) {
171 perror("fork failed");
172 return EXIT_FAILURE;
173 }
174
175 if (child_pid == 0) { // Child process
176 close(stdout_pipe[0]); // Close unused read end of stdout pipe
177 close(stderr_pipe[0]); // Close unused read end of stderr pipe
178
179 // Redirect stdout to stdout_pipe
180 if (dup2(stdout_pipe[1], STDOUT_FILENO) == -1) {
181 perror("dup2 stdout failed");
182 exit(EXIT_FAILURE);
183 }
184
185 // Redirect stderr to stderr_pipe
186 if (dup2(stderr_pipe[1], STDERR_FILENO) == -1) {
187 perror("dup2 stderr failed");
188 exit(EXIT_FAILURE);
189 }
190
191 close(stdout_pipe[1]); // Close the original write end of stdout pipe
192 close(stderr_pipe[1]); // Close the original write end of stderr pipe
193
194 // Execute the specified command
195 execve(argv[command_index], argv + command_index, envp);
196 perror("execve failed");
197 exit(EXIT_FAILURE);
198 }
199
200 // Parent process
201 close(stdout_pipe[1]); // Close unused write end of stdout pipe
202 close(stderr_pipe[1]); // Close unused write end of stderr pipe
203
204 // Create and set up timerfd for handling timeouts
205 int timer_fd = create_timer_fd(timeout);
206
207 // Create and set up signalfd for handling SIGCHLD
208 int signal_fd = create_signalfd();
209
210 // Set up poll file descriptors
211 struct pollfd poll_fds[4];
212 poll_fds[0].fd = signal_fd; // File descriptor for signal handling
213 poll_fds[0].events = POLLIN; // Monitor for readability
214 poll_fds[1].fd = timer_fd; // File descriptor for timer
215 poll_fds[1].events = POLLIN; // Monitor for readability
216 poll_fds[2].fd = stdout_pipe[0]; // File descriptor for stdout pipe
217 poll_fds[2].events = POLLIN; // Monitor for readability
218 poll_fds[3].fd = stderr_pipe[0]; // File descriptor for stderr pipe
219 poll_fds[3].events = POLLIN; // Monitor for readability
220
221 char buffer[1024]; // Buffer for reading pipe data
222 int child_status = 0; // Child exit status
223 int child_terminated = 0; // Flag to track if the child process has terminated
224
225 // Main event loop
226 while (1) {
227 // Wait for events on the file descriptors
228 int poll_result = poll(poll_fds, 4, -1);
229 if (poll_result == -1) {
230 perror("poll failed");
231 exit(EXIT_FAILURE);
232 }
233
234 // Handle SIGCHLD event from signalfd
235 if (poll_fds[0].revents & POLLIN) {
236 struct signalfd_siginfo siginfo;
237 if (read(signal_fd, &siginfo, sizeof(siginfo)) != sizeof(siginfo)) {
238 perror("read from signalfd failed");
239 exit(EXIT_FAILURE);
240 }
241 // SIGCHLD is received when the child process terminates
242 if (siginfo.ssi_signo == SIGCHLD) {
243 // Check the status of the child process
244 if (waitpid(siginfo.ssi_pid, &child_status, WNOHANG) != child_pid) {
245 perror("waitpid returns incorrect pid");
246 exit(EXIT_FAILURE);
247 }
248 // Mark that the child process is terminated
249 child_terminated = 1;
250 }
251 }
252
253 // Handle timeout event from timerfd
254 if (poll_fds[1].revents & POLLIN) {
255 uint64_t expirations = 0;
256 if (read(timer_fd, &expirations, sizeof(expirations)) == -1) {
257 perror("read from timerfd failed");
258 exit(EXIT_FAILURE);
259 }
260 // Timeout event indicates that the specified timeout period has expired
261 if (!child_terminated) {
262 // Forcefully kill the child process if not yet terminated
263 kill(child_pid, SIGKILL);
264 }
265 }
266
267 // Handle stdout pipe events
268 if (poll_fds[2].revents & POLLIN) {
269 ssize_t bytes_read = 0;
270
271 // Continue reading until no more data is available
272 while ((bytes_read = read(poll_fds[2].fd, buffer, sizeof(buffer))) > 0) {
273 write(STDOUT_FILENO, buffer, bytes_read);
274 }
275
276 if (bytes_read == 0) { // EOF
277 // The pipe is closed, which means the child process has finished writing
278 // Mark the file descriptor as invalid to stop polling it
279 poll_fds[2].fd = -1;
280 poll_fds[2].events = 0;
281 poll_fds[2].revents = 0;
282 } else { // error
283 if (errno != EAGAIN && errno != EWOULDBLOCK) {
284 perror("read from stdout pipe failed");
285 // Exit loop on fatal error
286 break;
287 }
288 }
289 }
290
291 // Handle stderr pipe events
292 if (poll_fds[3].revents & POLLIN) {
293 ssize_t bytes_read = 0;
294
295 // Continue reading until no more data is available
296 while ((bytes_read = read(poll_fds[3].fd, buffer, sizeof(buffer))) > 0) {
297 write(STDOUT_FILENO, buffer, bytes_read);
298 }
299
300 if (bytes_read == 0) { // EOF
301 // The pipe is closed, which means the child process has finished writing
302 // Mark the file descriptor as invalid to stop polling it
303 poll_fds[3].fd = -1;
304 poll_fds[3].events = 0;
305 poll_fds[3].revents = 0;
306 } else { // errors
307 if (errno != EAGAIN && errno != EWOULDBLOCK) {
308 perror("read from stderr pipe failed");
309 // Exit loop on fatal error
310 break;
311 }
312 }
313 }
314
315 // Break the loop if the child has terminated and no more data to read from both pipes
316 if (child_terminated && !(poll_fds[2].revents & POLLIN) && !(poll_fds[3].revents & POLLIN)) {
317 break; // Exit the loop if the child has terminated and no more data to read
318 }
319 }
320
321 // Close all open file descriptors
322 close(stdout_pipe[0]); // Close read end of stdout pipe
323 close(stderr_pipe[0]); // Close read end of stderr pipe
324 close(timer_fd); // Close timer file descriptor
325 close(signal_fd); // Close signalfd file descriptor
326
327 return EXIT_SUCCESS; // Exit the program successfully
328}
